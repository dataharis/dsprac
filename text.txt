        function submitQuiz() {
            let questions = document.querySelectorAll('.question-box');
            let score = 0;
            let total = questions.length;

            questions.forEach((q, index) => {
                let qName = 'q' + (index + 1);
                let selected = document.querySelector(`input[name="${qName}"]:checked`);
                let correctOpt = q.getAttribute('data-correct');
                let feedbackSpan = document.getElementById(`feedback-${qName}`);

                if (selected) {
                    if (selected.value === correctOpt) {
                        score++;
                        feedbackSpan.innerHTML = "✅ Correct!";
                        feedbackSpan.className = "correct-ans";
                    } else {
                        feedbackSpan.innerHTML = `❌ Incorrect. Correct answer was (${correctOpt}).`;
                        feedbackSpan.className = "incorrect-ans";
                    }
                } else {
                    feedbackSpan.innerHTML = `⚠️ Not answered. Correct answer was (${correctOpt}).`;
                    feedbackSpan.className = "incorrect-ans";
                }
            });

            let resultBox = document.getElementById('resultBox');
            resultBox.style.display = "block";
            let percentage = (score / total) * 100;
            resultBox.innerHTML = `You scored ${score} out of ${total} (${percentage}%).`;
            resultBox.style.background = percentage >= 70 ? "#dcfce7" : "#fee2e2";
            resultBox.style.color = percentage >= 70 ? "#166534" : "#991b1b";
        }
